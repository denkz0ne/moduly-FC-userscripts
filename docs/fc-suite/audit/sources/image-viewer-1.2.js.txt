// ==UserScript==
// @name         FaxCopy – interný prehliadač obrázkov
// @namespace    mato-faxcopy-image-viewer
// @version      1.2
// @description  Obrázkové odkazy otvorí v internom prehliadači namiesto stiahnutia.
// @match        *://moduly.faxcopy.sk/*
// @match        *://*.moduly.faxcopy.sk/*
// @grant        GM_xmlhttpRequest
// @connect      *
// @run-at       document-start
// ==/UserScript==

(() => {
    'use strict';

    const IMAGE_EXTENSIONS =
        /\.(?:avif|bmp|gif|jpe?g|png|svg|webp)(?:$|[?#])/i;

    let overlay = null;
    let objectUrl = null;
    let currentBlob = null;
    let currentUrl = '';
    let zoom = 1;

    function createButton(text) {
        const button = document.createElement('button');

        button.type = 'button';
        button.textContent = text;

        Object.assign(button.style, {
            border: '1px solid rgba(255,255,255,.25)',
            borderRadius: '8px',
            padding: '8px 13px',
            background: 'rgba(255,255,255,.12)',
            color: '#fff',
            cursor: 'pointer',
            font: '14px system-ui, sans-serif'
        });

        button.addEventListener('mouseenter', () => {
            button.style.background = 'rgba(255,255,255,.22)';
        });

        button.addEventListener('mouseleave', () => {
            button.style.background = 'rgba(255,255,255,.12)';
        });

        return button;
    }

    function getFilename(url, mimeType = '') {
        let filename = '';

        try {
            const parsedUrl = new URL(url);
            filename = decodeURIComponent(
                parsedUrl.pathname.split('/').pop() || ''
            );
        } catch {
            filename = '';
        }

        filename = filename.replace(/[\\/:*?"<>|]/g, '_');

        if (filename && /\.[a-z0-9]{2,5}$/i.test(filename)) {
            return filename;
        }

        const extensions = {
            'image/jpeg': 'jpg',
            'image/png': 'png',
            'image/gif': 'gif',
            'image/webp': 'webp',
            'image/avif': 'avif',
            'image/svg+xml': 'svg',
            'image/bmp': 'bmp'
        };

        return `obrazok.${extensions[mimeType] || 'jpg'}`;
    }

    function closeViewer() {
        if (objectUrl) {
            URL.revokeObjectURL(objectUrl);
            objectUrl = null;
        }

        overlay?.remove();

        overlay = null;
        currentBlob = null;
        currentUrl = '';
        zoom = 1;
    }

    function openViewer(url) {
        closeViewer();

        currentUrl = url;

        overlay = document.createElement('div');
        overlay.id = 'faxcopy-image-viewer';

        Object.assign(overlay.style, {
            position: 'fixed',
            inset: '0',
            zIndex: '2147483647',
            display: 'flex',
            flexDirection: 'column',
            background: 'rgba(10,10,12,.95)',
            backdropFilter: 'blur(8px)'
        });

        const toolbar = document.createElement('div');

        Object.assign(toolbar.style, {
            minHeight: '54px',
            display: 'flex',
            alignItems: 'center',
            gap: '8px',
            padding: '8px 12px',
            background: 'rgba(0,0,0,.5)',
            borderBottom: '1px solid rgba(255,255,255,.15)'
        });

        const info = document.createElement('div');
        info.textContent = 'Načítavam obrázok…';

        Object.assign(info.style, {
            flex: '1',
            overflow: 'hidden',
            textOverflow: 'ellipsis',
            whiteSpace: 'nowrap',
            color: '#fff',
            font: '14px system-ui, sans-serif'
        });

        const zoomOutButton = createButton('−');
        const zoomResetButton = createButton('100 %');
        const zoomInButton = createButton('+');
        const copyButton = createButton('Kopírovať URL');
        const downloadButton = createButton('Prevziať');
        const closeButton = createButton('✕');

        toolbar.append(
            info,
            zoomOutButton,
            zoomResetButton,
            zoomInButton,
            copyButton,
            downloadButton,
            closeButton
        );

        const stage = document.createElement('div');

        Object.assign(stage.style, {
            flex: '1',
            overflow: 'auto',
            display: 'grid',
            placeItems: 'center',
            padding: '20px'
        });

        const image = document.createElement('img');

        Object.assign(image.style, {
            display: 'block',
            maxWidth: 'calc(100vw - 40px)',
            maxHeight: 'calc(100vh - 95px)',
            objectFit: 'contain',
            transformOrigin: 'center center',
            transition: 'transform .08s ease',
            boxShadow: '0 12px 50px rgba(0,0,0,.55)'
        });

        stage.append(image);
        overlay.append(toolbar, stage);

        (document.body || document.documentElement).append(overlay);

        function updateZoom() {
            image.style.transform = `scale(${zoom})`;
            zoomResetButton.textContent = `${Math.round(zoom * 100)} %`;
        }

        zoomOutButton.addEventListener('click', () => {
            zoom = Math.max(0.1, zoom - 0.1);
            updateZoom();
        });

        zoomInButton.addEventListener('click', () => {
            zoom = Math.min(8, zoom + 0.1);
            updateZoom();
        });

        zoomResetButton.addEventListener('click', () => {
            zoom = 1;
            updateZoom();
        });

        copyButton.addEventListener('click', async () => {
            try {
                await navigator.clipboard.writeText(currentUrl);

                copyButton.textContent = 'Skopírované ✓';

                setTimeout(() => {
                    copyButton.textContent = 'Kopírovať URL';
                }, 1200);
            } catch {
                window.prompt('Adresa obrázka:', currentUrl);
            }
        });

        downloadButton.addEventListener('click', () => {
            if (!objectUrl || !currentBlob) {
                return;
            }

            const downloadLink = document.createElement('a');

            downloadLink.href = objectUrl;
            downloadLink.download = getFilename(
                currentUrl,
                currentBlob.type
            );

            document.body.append(downloadLink);
            downloadLink.click();
            downloadLink.remove();
        });

        closeButton.addEventListener('click', closeViewer);

        stage.addEventListener(
            'wheel',
            event => {
                if (!objectUrl) {
                    return;
                }

                event.preventDefault();

                zoom += event.deltaY < 0 ? 0.1 : -0.1;
                zoom = Math.min(8, Math.max(0.1, zoom));

                updateZoom();
            },
            { passive: false }
        );

        GM_xmlhttpRequest({
            method: 'GET',
            url,
            responseType: 'blob',

            onload(response) {
                if (response.status < 200 || response.status >= 400) {
                    info.textContent = `Chyba HTTP ${response.status}`;
                    return;
                }

                currentBlob = response.response;
                objectUrl = URL.createObjectURL(currentBlob);

                image.src = objectUrl;

                image.addEventListener('load', () => {
                    info.textContent =
                        `${getFilename(currentUrl, currentBlob.type)} — ` +
                        `${image.naturalWidth} × ${image.naturalHeight} px`;
                });

                image.addEventListener('error', () => {
                    info.textContent =
                        'Súbor sa nepodarilo rozpoznať ako obrázok.';
                });
            },

            onerror() {
                info.textContent =
                    'Obrázok sa nepodarilo načítať.';
            }
        });
    }

    document.addEventListener(
        'click',
        event => {
            if (event.button !== 0) {
                return;
            }

            const link = event.target.closest('a[href]');

            if (!link) {
                return;
            }

            const url = link.href;
            const looksLikeImage = IMAGE_EXTENSIONS.test(url);

            // Alt + klik skúsi ako obrázok otvoriť aj link bez prípony.
            if (!looksLikeImage && !event.altKey) {
                return;
            }

            // Ctrl, Shift alebo Cmd ponechá klasické správanie Chrome.
            if (
                (event.ctrlKey || event.metaKey || event.shiftKey) &&
                !event.altKey
            ) {
                return;
            }

            event.preventDefault();
            event.stopPropagation();
            event.stopImmediatePropagation();

            openViewer(url);
        },
        true
    );

    document.addEventListener('keydown', event => {
        if (event.key === 'Escape' && overlay) {
            closeViewer();
        }
    });
})();
